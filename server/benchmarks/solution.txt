#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;

    vector<int> nums(n);
    for (int i = 0; i < n; i++) cin >> nums[i];

    int target;
    cin >> target;

    unordered_map<int, int> mp;

    for (int i = 0; i < n; i++) {
        int need = target - nums[i];
        if (mp.count(need)) {
            int a = mp[need], b = i;
            if (a > b) swap(a, b);
            cout << a << " " << b;
            break;
        }
        mp[nums[i]] = i;
    }
    return 0;
}
